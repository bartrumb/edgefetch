#include <tunables/global>

profile edgefetch_fat flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/ssl_certs>
  #include <abstractions/fonts>

  # Only outbound TCP/UDP (DNS + HTTPS/WSS + loopback CDP/SSRFProxy); explicitly deny raw/packet sockets.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  network unix dgram,
  unix (create, bind, listen, accept, connect, send, receive, getattr, setattr, setopt, getopt, shutdown),
  # Unrestricted peer label so signals continue working after Home Assistant Supervisor rewrites
  # `profile edgefetch_fat` -> `profile <repo_slug>_edgefetch_fat` via `adjust_profile()`.
  signal (send, receive),
  deny network raw,
  deny network packet,

  # Needed once at millisecond 0 (while UID == 0) so `priv.LockDownSecretPaths` enforces `0700 /data` +
  # `0600 /data/options.json` (`root:root`) right before `priv.DropRoot()` permanently drops all threads
  # to `65532:65532` and clears all capabilities (`PR_SET_NO_NEW_PRIVS=1`).
  capability setuid,
  capability setgid,
  capability chown,
  capability fowner,

  # Explicitly deny dangerous container-escape capabilities and kernel interfaces
  deny capability dac_override,
  deny capability dac_read_search,
  deny capability sys_admin,
  deny capability sys_ptrace,
  deny capability sys_module,
  deny capability sys_rawio,
  deny capability net_admin,
  deny mount,
  deny remount,
  deny umount,
  deny ptrace,

  # Binary, Chromium (`mrix` compatible with AppArmor 3.x on Home Assistant OS + `PR_SET_NO_NEW_PRIVS`),
  # Poppler `pdftotext`, and pre-drop `/data/options.json` lockdown access
  / r,
  /edgefetch ixr,
  /usr/ r,
  /usr/bin/ r,
  /usr/bin/** ixr,
  /usr/lib/ r,
  /usr/lib/** mr,
  /usr/lib/chromium/ r,
  /usr/lib/chromium/** mrix,
  /lib/ r,
  /lib/** mr,
  /usr/share/ r,
  /usr/share/** r,
  /var/cache/fontconfig/ rw,
  /var/cache/fontconfig/** rwkl,
  /etc/ r,
  /etc/** r,
  /data/ rw,
  /data/options.json rw,
  /tmp/ rw,
  /tmp/** rwklm,
  /dev/ r,
  /dev/shm/ rw,
  /dev/shm/** rwklm,
  /memfd:* rwklm,
  /memfd:** rwklm,
  /SYSV* rwklm,
  /dev/null rw,
  /dev/zero rw,
  /dev/urandom rw,
  /dev/random r,
  /proc/ r,
  /proc/** r,
  /proc/@{pid}/oom_score_adj rw,
  /proc/@{pid}/clear_refs w,
  /proc/@{pid}/task/*/comm rw,
  /proc/@{pid}/fd/ r,
  /sys/ r,
  /sys/fs/cgroup/ r,
  /sys/fs/cgroup/** r,
  /sys/bus/ r,
  /sys/bus/** r,
  /sys/class/ r,
  /sys/class/** r,
  /sys/devices/ r,
  /sys/devices/** r,

  # Block sensitive kernel / hardware paths
  deny /proc/kcore rwklx,
  deny /proc/kmem rwklx,
  deny /proc/mem rwklx,
  deny /sys/firmware/** rwklx,
}

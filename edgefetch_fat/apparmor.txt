#include <tunables/global>

profile edgefetch_fat flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/ssl_certs>
  #include <abstractions/fonts>

  # Only outbound TCP/UDP (DNS + HTTPS/WSS + loopback CDP/SSRFProxy); explicitly deny raw/packet sockets.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  network unix dgram,
  deny network raw,
  deny network packet,

  # Needed once at startup before permanent privilege drop (`internal/priv/priv_linux.go`)
  capability setuid,
  capability setgid,

  # Explicitly deny dangerous container-escape capabilities and kernel interfaces
  deny capability sys_admin,
  deny capability sys_ptrace,
  deny capability sys_module,
  deny capability sys_rawio,
  deny capability net_admin,
  deny mount,
  deny remount,
  deny umount,
  deny ptrace,

  # Binary, Chromium, Poppler pdftotext & read-only config access
  /edgefetch ixr,
  /usr/bin/** ixr,
  /usr/lib/** mr,
  # Chromium's binary and its helpers (chrome_crashpad_handler) live here, not in /usr/bin. No shell is allowed.
  /usr/lib/chromium/** mrix,
  /lib/** mr,
  # Read-only package data. Chromium aborts at start without ICU's data in /usr/share/icu (U_FILE_ACCESS_ERROR, 2.0.1).
  /usr/share/** r,
  /var/cache/fontconfig/** rwk,
  /etc/** r,
  /data/ r,
  /data/options.json r,
  /tmp/** rwkl,
  /dev/shm/** rwkl,
  /dev/null rw,
  /dev/urandom r,
  /dev/random r,
  /proc/** r,
  /sys/fs/cgroup/** r,
  /sys/devices/system/cpu/** r,

  # Block sensitive kernel / hardware paths
  deny /proc/kcore rwklx,
  deny /proc/kmem rwklx,
  deny /proc/mem rwklx,
  deny /sys/firmware/** rwklx,
}
